cmake_minimum_required(VERSION 3.22.1)
project(speakup_dsp LANGUAGES CXX)

add_library(speakup_frame STATIC frame_engine.cpp)
target_include_directories(speakup_frame PUBLIC ${CMAKE_CURRENT_LIST_DIR})
target_compile_features(speakup_frame PUBLIC cxx_std_17)
set_target_properties(speakup_frame PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(speakup_frame PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)

if(ANDROID)
  add_library(speakup_dsp SHARED jni_bridge.cpp)
  target_link_libraries(speakup_dsp PRIVATE speakup_frame log)
  target_compile_features(speakup_dsp PRIVATE cxx_std_17)
  target_compile_options(speakup_dsp PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
endif()

include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
  add_executable(frame_engine_test ../../../../../tests/native/frame_engine_test.cpp)
  target_link_libraries(frame_engine_test PRIVATE speakup_frame)
  target_compile_features(frame_engine_test PRIVATE cxx_std_17)
  target_compile_options(frame_engine_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME frame_engine_contract COMMAND frame_engine_test)
endif()
